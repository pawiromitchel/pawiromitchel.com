29:T51c,import * as cheerio from "cheerio";

interface BankRate {
  bank: string;
  currency: "USD" | "EUR";
  buyCash: number;
  sellCash: number;
  buyWire: number;
  sellWire: number;
}

export function parseBankHTML(html: string, bankName: string): BankRate[] {
  const $ = cheerio.load(html);
  const rates: BankRate[] = [];

  $("tr").each((_, element) => {
    const rowText = $(element).text().toUpperCase();

    const isUSD = rowText.includes("USD") || rowText.includes("DOLLAR");
    const isEUR = rowText.includes("EUR") || rowText.includes("EURO");

    if (!isUSD && !isEUR) return;

    // Grab all numerical values from this row, normalizing commas to decimals
    const values = $(element)
      .find("td")
      .map((_, col) => $(col).text().trim().replace(",", "."))
      .get()
      .map((text) => parseFloat(text))
      .filter((num) => !isNaN(num) && num > 10 && num < 100); // Sanity check range

    // If we extracted at least Buy and Sell numbers
    if (values.length >= 2) {
      rates.push({
        bank: bankName,
        currency: isUSD ? "USD" : "EUR",
        buyCash: values[0],
        sellCash: values[1],
        buyWire: values[2] ?? values[0], // fallback if cash and wire are identical
        sellWire: values[3] ?? values[1],
      });
    }
  });

  return rates;
}
